        // Initial Coordinates
        const BASE_X = 300;
        const BASE_Y = 200; // Starting high in the pelvis

        // --- Functions ---

        function init() {
            // Generate Buttons
            steps.forEach((step, index) => {
                const btn = document.createElement('button');
                btn.className = `step-card px-3 py-1 text-xs rounded-full border border-slate-700 bg-slate-800 text-slate-300 hover:bg-slate-700 transition ${index === 0 ? 'active' : ''}`;
                btn.innerText = step.name;
                btn.onclick = () => jumpToStep(index);
                stepButtonsContainer.appendChild(btn);
            });

            updateVisuals(0);
            setupEventListeners();
        }

        function setupEventListeners() {
            timeline.addEventListener('input', (e) => {
                const val = parseFloat(e.target.value);
                // Determine step based on value
                const stepIndex = Math.round(val);
                if (currentStep !== stepIndex) {
                    jumpToStep(val); // Pass float for smooth interpolation if we wanted, but rounding is cleaner for discrete steps
                }
            });

            playPauseBtn.addEventListener('click', togglePlay);
            
            document.getElementById('resetBtn').addEventListener('click', () => {
                stopAutoPlay();
                jumpToStep(0);
            });
        }

        function togglePlay() {
            isPlaying = !isPlaying;
            if (isPlaying) {
                playIcon.classList.add('hidden');
                pauseIcon.classList.remove('hidden');
                startAutoPlay();
            } else {
                playIcon.classList.remove('hidden');
                pauseIcon.classList.add('hidden');
                stopAutoPlay();
            }
        }

        function startAutoPlay() {
            animationInterval = setInterval(() => {
                if (currentStep < steps.length - 1) {
                    jumpToStep(currentStep + 1);
                } else {
                    togglePlay(); // Stop at end
                }
            }, 2500); // 2.5 seconds per step
        }

        function stopAutoPlay() {
            clearInterval(animationInterval);
            isPlaying = false;
            playIcon.classList.remove('hidden');
            pauseIcon.classList.add('hidden');
        }

        function jumpToStep(index) {
            // Clamp index
            if (index < 0) index = 0;
            if (index > steps.length - 1) index = steps.length - 1;

            currentStep = index;
            timeline.value = index;
            updateVisuals(index);
            updateContent(index);
        }

        function updateContent(index) {
            const stepData = steps[index];
            
            // Animate Text Fade
            const contentDiv = document.getElementById('content-display');
            contentDiv.style.opacity = '0.5';
            
            setTimeout(() => {
                stepTitle.innerText = stepData.name;
                stepDesc.innerText = stepData.desc;
                structureDetail.innerText = stepData.structure;
                movementType.innerText = stepData.movement;
                clinicalPearl.innerText = stepData.pearl;
                stepIndicator.innerText = stepData.name;
                contentDiv.style.opacity = '1';
            }, 100);

            // Update Buttons
            const btns = stepButtonsContainer.children;
            for (let i = 0; i < btns.length; i++) {
                if (i === index) {
                    btns[i].classList.add('active');
                    btns[i].scrollIntoView({ behavior: 'smooth', block: 'nearest' });
                } else {
                    btns[i].classList.remove('active');
                }
            }
        }

        function updateVisuals(index) {
            const stepData = steps[index];
            
            // 1. Move Head Group (Position)
            // We add the transform values to the base coordinates
            // Note: SVG Y is down, so +Y is down (descent)
            // However, the pelvis drawing is centered at 300,300.
            // Let's assume Base Y 200 is above the pelvic inlet.
            
            // To make it fit the drawn pelvis better:
            // Pelvis inlet is around y=-100 relative to group (300,300) -> global y=200.
            // Let's adjust base positions to match the visual drawing.
            
            // Base SVG coordinates:
            // Head Center Start: (300, 200)
            // Step 0 (Engagement): Just entering. (300, 220)
            // Step 4 (Extension): Under pubis. (300, 300 + 130 = 430)
            
            const targetX = BASE_X + stepData.transform.x;
            const targetY = BASE_Y + stepData.transform.y;
            
            headGroup.style.transform = `translate(${targetX}px, ${targetY}px)`;

            // 2. Rotate Head (Orientation)
            // The inner wrapper rotates to simulate flexion/internal rotation
            // 0 = Occiput Up (Vertex presentation transverse)
            // 90 = Occiput Right (Transverse)
            // 130 = Occiput Anterior (Internal Rotation complete) -> Face is posterior
            // Actually, let's map rotation to Occiput position.
            // 0 deg: Occiput at 12 o'clock.
            // 90 deg: Occiput at 9 o'clock.
            // 130 deg: Occiput at ~7 o'clock (Anterior).
            
            const rotation = stepData.transform.rot;
            headRotationWrapper.style.transform = `rotate(${rotation}deg)`;

            // 3. Update Mini Map
            miniHead.setAttribute('cx', 100 + (stepData.transform.x / 2)); // Scale down movement
            miniHead.setAttribute('cy', 50 + (stepData.transform.y / 3));
            // Rotate mini head based on internal rotation (90 deg rotation = mini head 90 deg)
            let miniR = stepData.miniRot;
            if(stepData.name === "Expulsion") {
                 miniR += 45; // Tilt for delivery
                 // Scale to tear drop for expulsion?
            }
            miniHead.setAttribute('transform', `rotate(${miniR}, ${100 + (stepData.transform.x / 2)}, ${50 + (stepData.transform.y / 3)})`);


            // 4. Update Metrics Text
            // Calculate Station (Approximate logic for visualizer)
            // Ischial spines are at local(0,0) of pelvis group which is Global(300, 300).
            // Head starts at Global(300, 200).
            // Distance to station 0: 100px.
            // Let's say 100px = 5cm station? That makes 1cm = 20px.
            // Current Y center of head.
            const currentHeadY = targetY;
            const stationZeroY = 300; 
            const diffY = stationZeroY - currentHeadY; // Positive if above (negative station)
            
            // Mapping: 
            // Start (200) -> Diff 100 -> Station -5.
            // Station 0 (300) -> Diff 0 -> Station 0.
            // Extension (330) -> Diff -30 -> Station +1.5.
            
            // Let's simplify: 
            // Diff / 20 = Station.
            let stationVal = (diffY / 20); 
            // Clamp for display
            if(stationVal < 0) stationVal = Math.ceil(stationVal * 10) / 10;
            else stationVal = Math.floor(stationVal * 10) / 10;
            
            stationDisplay.innerText = stationVal.toFixed(1);
            rotationDisplay.innerText = rotation + "°";
        }

                });
                el.addEventListener('mouseleave', () => {
                    tooltip.style.opacity = '0';
                });
            }
        });
